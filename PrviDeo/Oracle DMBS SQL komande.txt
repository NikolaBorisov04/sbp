CREATE TABLE TipPogona (

    naziv VARCHAR(50) NOT NULL UNIQUE -- 'Elektricno', 'Hibridno', 'Klasicno'
);

CREATE TABLE TipKoriscenja (
    id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    naziv VARCHAR(50) NOT NULL UNIQUE -- 'Putnicko', 'Dostavno', 'Specijalno'
);

CREATE TABLE Vozilo (
    id NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    registarska_oznaka VARCHAR2(20) NOT NULL UNIQUE,
    vin VARCHAR2(17) NOT NULL UNIQUE,
    marka VARCHAR2(50) NOT NULL,
    model VARCHAR2(50) NOT NULL,
    godina_proizvodnje NUMBER(4) NOT NULL,
    datum_nabavke DATE NOT NULL,
    status VARCHAR2(50) NOT NULL,
    broj_sedista NUMBER NOT NULL,
    stanje_enterijera VARCHAR2(1000),
    stanje_eksterijera VARCHAR2(1000),
    ogranicenja_koriscenja VARCHAR2(2000),
    
    tip_koriscenja_id NUMBER NOT NULL,
    tip_pogona_id NUMBER NOT NULL,

    CONSTRAINT fk_vozilo_tip_koriscenja FOREIGN KEY (tip_koriscenja_id) REFERENCES TipKoriscenja(id),
    CONSTRAINT fk_vozilo_tip_pogon FOREIGN KEY (tip_pogona_id) REFERENCES TipPogona(id)
);

CREATE TABLE ElektricnoVozilo (
    vozilo_id NUMBER PRIMARY KEY,
    kapacitet_baterije NUMBER(8, 2) NOT NULL,
    trenutni_nivo_napunjenosti NUMBER(5, 2) NOT NULL,

    CONSTRAINT fk_el_vozilo FOREIGN KEY (vozilo_id) REFERENCES Vozilo(id) ON DELETE CASCADE

CREATE TABLE HibridnoVozilo (

CREATE TABLE KlasicnoVozilo (

    zapremina_rezervoara NUMBER(8, 2) NOT NULL,

CREATE TABLE Oprema (

    opis VARCHAR2(1000)

CREATE TABLE VoziloOprema (
